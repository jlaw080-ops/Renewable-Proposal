# Phase 0 — 기반 구축 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
> 마스터플랜: `docs/superpowers/plans/2026.07.26_전면개편_마스터플랜.md` (Global Constraints 포함)

**Goal:** Next.js 스캐폴드 + 레포 재구성 + 엔진 무수정 로드 체계를 구축하고, 개편 전후 계산 결과 동일성을 스냅샷으로 증명 가능하게 만든다.

**Architecture:** 작업 브랜치 `revamp/next`. 런타임 스크립트 모듈은 `public/`으로 git mv(내용 무수정), ES 모듈 `engine/`은 루트 유지 후 직접 import. 클라이언트 어댑터 훅이 스크립트를 순서대로 주입해 `window.LIB_*` 전역을 구성한다.

**Tech Stack:** Next.js(App Router)·React·JavaScript·Node(verify 하니스)

- [ ] **Step 2: 로더 훅 작성** — `lib/useEngineReady.js`

```js
"use client";
import { useEffect, useState } from "react";
import { ENGINE_SCRIPTS, READY_GLOBALS } from "./engineScripts";

let loadPromise = null;

function loadScript(src) {
  return new Promise((resolve, reject) => {
    const s = document.createElement("script");
    s.src = src;
    s.async = false;
    s.onload = resolve;
    s.onerror = () => reject(new Error("스크립트 로드 실패: " + src));
    document.head.appendChild(s);
  });
}

export function loadEngine() {
  if (!loadPromise) {
    loadPromise = ENGINE_SCRIPTS
      .reduce((p, src) => p.then(() => loadScript(src)), Promise.resolve())
      .then(() => {
        const missing = READY_GLOBALS.filter(g => !(g in window));
        if (missing.length) throw new Error("엔진 전역 누락: " + missing.join(", "));
      });
  }
  return loadPromise;
}

export function useEngineReady() {
  const [state, setState] = useState({ ready: false, error: null });
  useEffect(() => {
    let alive = true;
    loadEngine()
      .then(() => alive && setState({ ready: true, error: null }))
      .catch(e => alive && setState({ ready: false, error: e.message }));
    return () => { alive = false; };
  }, []);
  return state;
}
```

"use client";
import { useEffect, useState } from "react";
import { useEngineReady } from "@/lib/useEngineReady";

export default function EngineCheck() {
  const { ready, error } = useEngineReady();
  const [result, setResult] = useState(null);
  const [runError, setRunError] = useState(null);

  useEffect(() => {
    if (!ready) return;
    (async () => {
      try {
        const [{ runCalculation }, input1, input2] = await Promise.all([
          import("@/engine/index.js"),
          fetch("/fixtures/Input1-사업정보.json").then(r => r.json()),
          fetch("/fixtures/Input2-시나리오정보.json").then(r => r.json()),
        ]);
        setResult(await runCalculation(input1, input2, "가"));
      } catch (e) { setRunError(e.message); }
    })();
  }, [ready]);

  if (error || runError) return <pre data-testid="engine-error">ERROR: {error || runError}</pre>;
  if (!result) return <p>엔진 로딩 중…</p>;
  return <pre data-testid="engine-result">{JSON.stringify(result, null, 2)}</pre>;
}

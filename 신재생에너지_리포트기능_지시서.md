> **전제 조건**: 본 지시서는 기존 웹앱(`index.html`, `engine/`, `review/`)이 완성된 상태에서
> 리포트 생성 및 다운로드 기능을 추가하는 작업을 위한 지시서입니다.
> 기존 파일을 수정할 때는 반드시 기존 코드를 확인한 후 최소한으로 변경하세요.

```javascript
function collectReportData() {
  if (!window.LAST_OUTPUT1 || !window.LAST_OUTPUT2) {
    alert('계산 결과가 없습니다. 입력값을 먼저 채워주세요.');
    return null;
  }
  return {
    input1:      window.LAST_INPUT1,
    output1:     window.LAST_OUTPUT1,
    output2:     window.LAST_OUTPUT2,
    reviewHTML:  document.querySelector('.ql-editor')?.innerHTML ?? '',
    projectName: document.getElementById('input-project-name')?.value ?? '',

    logoImage:   window.LOGO_IMAGE_BASE64 ?? null,
  };
}
```

## 4. report/reportBuilder.js

각 페이지 HTML을 생성하는 순수 함수 모음. DOM 조작 없이 HTML 문자열만 반환합니다.

```javascript
// report/reportBuilder.js
// 리포트 HTML 생성 — DOM 조작 없이 문자열만 반환

'use strict';

// ── 유틸 ────────────────────────────────────────────
function fmt(n, digits = 0) {
  if (n == null || n === '') return '-';
  const num = parseFloat(n);
  if (isNaN(num)) return n;
  return num.toLocaleString('ko-KR', {
    minimumFractionDigits: digits,
    maximumFractionDigits: digits,
  });
}

function today() {
  return new Date().toLocaleDateString('ko-KR', {
    year: 'numeric', month: '2-digit', day: '2-digit'
  });
}

function statusBadge(만족여부) {
  if (만족여부 === 'Yes') return '<span class="status-pass">✓ PASS</span>';
  if (만족여부 === 'No')  return '<span class="status-fail">✗ FAIL</span>';
  return '<span class="status-na">해당없음</span>';
}

// ── Page 1: 표지 ─────────────────────────────────────
function buildCoverPage(data) {
  const { projectName, coverImage, logoImage } = data;

  const coverImgHTML = coverImage
    ? `<img src="${coverImage}" alt="표지 이미지">`
    : `<span class="report-cover-image-empty">이미지 없음</span>`;

  const logoHTML = logoImage
    ? `<img class="report-cover-logo" src="${logoImage}" alt="로고">`
    : '';

  return `
    <div class="report-page report-cover">
      <div class="report-cover-main">신재생에너지 검토 보고서</div>
      <div class="report-cover-sub">${projectName || '&nbsp;'}</div>
      <div class="report-cover-image-wrap">${coverImgHTML}</div>
      <div class="report-cover-date">${today()}</div>
      ${logoHTML}
    </div>`;
}

// ── Page 2: 목차 ─────────────────────────────────────
function buildTocPage() {
  const items = [
    { num: '1', name: '검토결과 요약', page: 3 },
    { num: '2', name: '사업개요',     page: 4 },
    { num: '3', name: '기준검토',     page: 5 },
    { num: '4', name: '설치 비율 검토', page: 6 },
  ];

  const rows = items.map(i => `
    <div class="toc-item">
      <span class="toc-item-num">${i.num}.</span>
      <span class="toc-item-name">${i.name}</span>
      <span class="toc-item-dots"></span>
      <span class="toc-item-page">${i.page}</span>
    </div>`).join('');

  return `
    <div class="report-page">
      ${header('목차', 2)}
      <div class="report-page-title">목  차</div>
      ${rows}
    </div>`;
}

// ── Page 3: 검토결과 요약 ─────────────────────────────
function buildResultPage(data) {
  const { output2, reviewHTML } = data;

  const summaryRows = output2.map(alt => `
    <tr>
      <td>${alt.id}</td>
      <td>${fmt(alt.비율, 1)} %</td>

      <td>${statusBadge(alt.만족여부)}</td>
    </tr>`).join('');

  return `
    <div class="report-page">
      ${header('검토결과', 3)}
      <div class="report-page-title">검토결과 요약</div>
      <div class="report-section-title">신재생에너지 적용 비율 검토 요약</div>
      <table class="report-table">
        <thead>
          <tr>
            <th>구분</th><th>설치 비율</th><th>의무 비율</th><th>만족 여부</th>
          </tr>
        </thead>

      <div class="report-section-title">검토 의견</div>
      <div class="review-text-wrap">${reviewHTML || '(검토 의견 없음)'}</div>

}

// ── Page 5: 기준검토 ──────────────────────────────────
function buildCriteriaPage(data) {
  const { input1 } = data;
  const 사업형태 = input1.사업형태;
  const 연도컬럼 = ['2024','2025','2026','2027','2028','2029','2030'];

  const 페이지제목 = 사업형태 === '공공'
    ? '공공기관 에너지이용 합리화 추진에 관한 규정'
    : `${input1.대지위치 || ''} 녹색건축물 설계기준`;

  let tableHTML = '';

  if (사업형태 === '공공') {
    // 공공: 단일 행
    const lib = window.LIB_의무비율?.['공공'] ?? {};
    const yearCells = 연도컬럼.map(y =>
      `<th>${y}</th>`).join('');
    const valueCells = 연도컬럼.map(y => {
      const v = lib[y];
      return `<td>${(v !== '' && v != null) ? v + ' %' : '-'}</td>`;
    }).join('');

    tableHTML = `
      <table class="report-table">
        <thead><tr><th>구분</th>${yearCells}</tr></thead>
        <tbody>
          <tr>
            <td style="font-weight:600">공공</td>${valueCells}
          </tr>
        </tbody>
      </table>`;

  } else {
    // 민간: 가/나/다/라 전체
    const lib = window.LIB_의무비율 ?? {};
    const 선택카테고리 = input1.카테고리;

    // 주거구분 판단
    const 주거구분 = (input1.용도별연면적목록 ?? [])
      .some(x => x.용도 !== '공동주택') ? '비주거' : '주거';

    // 해당 지역 + 주거구분 필터링
    const 지역키 = window.normalize대지위치?.(input1.대지위치) ?? '';
    const 후보 = Object.values(lib).filter(v =>
      v.state === 지역키 && v.purpose === 주거구분
    );
    const 카테고리순 = ['가','나','다','라'];
    const 정렬 = 카테고리순
      .map(c => 후보.find(v => v.category === c))
      .filter(Boolean);

    if (정렬.length === 0) {
      tableHTML = `<p style="color:#999; font-size:11px;">해당 지역 기준 데이터 없음</p>`;
    } else {
      const yearCells = 연도컬럼.map(y => `<th>${y}</th>`).join('');
      const rows = 정렬.map(item => {
        const isSelected = item.category === 선택카테고리;
        const valueCells = 연도컬럼.map(y => {
          const v = item[y];
          if (v === '' || v == null) return '<td>-</td>';
          return `<td>${v === '자율' ? '자율' : v + ' %'}</td>`;
        }).join('');
        return `
          <tr class="${isSelected ? 'highlight-row' : ''}">
            <td style="font-weight:600">${item.category}</td>
            <td>${item.녹색건축인증 || '-'}</td>
            <td>${item.에너지효율등급 || '-'}</td>

          </tr>`;
      }).join('');

      tableHTML = `
        <table class="report-table">
          <thead>
            <tr>
              <th>등급</th><th>녹색건축인증</th><th>에너지효율등급</th>${yearCells}
            </tr>
          </thead>
          <tbody>${rows}</tbody>
        </table>`;
    }
  }

  return `

      ${header('기준검토', 5)}
      <div class="report-page-title">${페이지제목}</div>
```

```javascript
'use strict';

function _filename(ext) {
  const d = new Date().toISOString().slice(0, 10);
  return `신재생에너지검토_${d}.${ext}`;

// ── PDF 다운로드 ────────────────────────────────────
async function downloadReportPDF(reportHTML) {
  const btn = document.getElementById('btn-report-pdf');
  const modalBtn = document.getElementById('modal-btn-pdf');
  [btn, modalBtn].forEach(b => b && (b.disabled = true, b.textContent = '변환 중...'));

  try {
    const { jsPDF } = window.jspdf;
    const pdf = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });

    // 임시 컨테이너에 렌더링
    const container = document.createElement('div');
    container.style.cssText = `
      position: absolute; left: -9999px; top: 0;
      width: 794px; background: white;
    `;
    // 스타일 포함
    const styleLink = document.createElement('link');
    styleLink.rel = 'stylesheet';
    styleLink.href = 'report/reportStyle.css';
    container.innerHTML = reportHTML;
    document.body.appendChild(container);

    // 폰트 로드 대기
    await document.fonts.ready;
    await new Promise(r => setTimeout(r, 400)); // 렌더링 대기

    const pages = container.querySelectorAll('.report-page');
    for (let i = 0; i < pages.length; i++) {
      const canvas = await html2canvas(pages[i], {
        scale: 2,
        useCORS: true,
        backgroundColor: '#ffffff',
        logging: false,
      });

      const imgData = canvas.toDataURL('image/png');
      const pdfW = pdf.internal.pageSize.getWidth();
      const pdfH = (canvas.height * pdfW) / canvas.width;

      if (i > 0) pdf.addPage();
      pdf.addImage(imgData, 'PNG', 0, 0, pdfW, Math.min(pdfH, 297));
    }

    document.body.removeChild(container);
    pdf.save(_filename('pdf'));

  } catch (err) {
    console.error('PDF 변환 오류:', err);
    alert('PDF 변환 중 오류가 발생했습니다: ' + err.message);
  } finally {
    if (btn)      { btn.disabled = false;      btn.textContent = 'PDF 다운로드'; }
    if (modalBtn) { modalBtn.disabled = false; modalBtn.textContent = 'PDF 다운로드'; }
  }
}

function downloadReportWord(reportHTML) {
  if (!window.htmlDocx) {
    alert('html-docx-js 라이브러리가 로드되지 않았습니다.');
    return;
  }
  // 외부 스타일시트를 인라인으로 교체 (Word는 외부 CSS 미지원)
  const inlined = reportHTML.replace(

  );
  const blob = window.htmlDocx.asBlob(inlined);
  _downloadBlob(blob, _filename('docx'),
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document');
}

// ── 공통 유틸 ───────────────────────────────────────
function _downloadBlob(content, filename, type) {
  const blob = (content instanceof Blob)
    ? content
    : new Blob([content], { type });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(a.href);
}
```

## 10. 엣지 케이스 처리 규칙

| 상황 | 처리 방법 |
|------|----------|
| 리포트 생성 시 계산 결과 없음 | "계산 결과가 없습니다. 입력값을 먼저 채워주세요" 경고 후 중단 |
| 표지 이미지 미업로드 | 이미지 영역을 점선 빈 박스로 표시, 리포트 생성은 계속 진행 |
| 로고 이미지 없음 | 로고 영역 생략 (에러 없이 처리) |
